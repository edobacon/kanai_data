**Pseudo-code dkc_enforce_step**:
```python
@mcp.tool()
def dkc_enforce_step(file_path: str, step_id: str | None = None, ticket_id: str | None = None) -> dict:
    md = Path(file_path).read_text()
    tag_re = r'<!-- enforcement: a-hook command: "([^"]+)" -->'
    commands = re.findall(tag_re, md)
    executed = []
    for cmd in commands:
        # Resolver placeholders {TICKET-id}, {SPEC-id}, {project}, {module}
        resolved = resolve_placeholders(cmd, ticket_id=ticket_id, step_id=step_id)
        result = subprocess.run(resolved, shell=True, capture_output=True, text=True, timeout=10)
        executed.append({
            "cmd": resolved,
            "exit_code": result.returncode,
            "stdout": result.stdout[:500],
            "stderr": result.stderr[:500],
        })
    return {"executed": executed, "log": f"{len(commands)} commands ejecutados"}
```

**Pseudo-code dkc_invoke_agent**:
```python
@mcp.tool()
def dkc_invoke_agent(subagent_type: str, prompt: str, tier: str | None = None, output_schema: str | None = None) -> dict:
    # Resolver via HOR-029 mapping en prompts/agent-tiers.md
    mapping = load_host_mapping()  # cache modulo-level
    resolved = mapping.get(subagent_type)
    if resolved == "—" or resolved is None:
        return {"executed": False, "mode": "inline-fallback",
                "message": f"Role '{subagent_type}' mapea a inline. Ejecutar manteniendo limites."}
    return {"executed": False, "mode": "agent",
            "message": f"Invoke Agent({subagent_type='{resolved}', prompt='{prompt}'})",
            "result": None}
    # NO spawn — el LLM principal invoca Agent tool localmente
```
